package com.pokewallet.crypto

import org.bouncycastle.crypto.digests.SHA256Digest
import org.bouncycastle.crypto.ec.CustomNamedCurves
import org.bouncycastle.crypto.params.ECDomainParameters
import org.bouncycastle.crypto.params.ECPrivateKeyParameters
import org.bouncycastle.crypto.signers.ECDSASigner
import org.bouncycastle.crypto.signers.HMacDSAKCalculator
import org.bouncycastle.crypto.signers.SchnorrSigner
import org.bouncycastle.jce.ECNamedCurveTable
import org.bouncycastle.jce.provider.BouncyCastleProvider
import org.bouncycastle.math.ec.ECPoint
import java.io.ByteArrayOutputStream
import java.math.BigInteger
import java.security.Security

/**
 * Secp256k1
 *
 * Funções:
 *  - Derivação de chave pública (comprimida)
 *  - Assinatura ECDSA determinística (RFC6979)
 *  - LOW-S enforcement (Bitcoin standard)
 *  - DER encoding estrito (BIP-66)
 *  - Assinatura Schnorr (BIP-340 / Taproot)
 */
object Secp256k1 {

    // =================================================
    // Provider (garantia)
    // =================================================

    init {
        if (Security.getProvider(BouncyCastleProvider.PROVIDER_NAME) == null) {
            Security.addProvider(BouncyCastleProvider())
        }
    }

    // =================================================
    // Parâmetros da curva (globais)
    // =================================================

    private val CURVE = CustomNamedCurves.getByName("secp256k1")

    private val DOMAIN = ECDomainParameters(
        CURVE.curve,
        CURVE.g,
        CURVE.n,
        CURVE.h
    )

    private val HALF_CURVE_ORDER: BigInteger =
        CURVE.n.shiftRight(1)

    // =================================================
    // Chave pública (EXISTENTE — PRESERVADA)
    // =================================================

    /**
     * Gera chave pública comprimida (33 bytes)
     *
     * Prefixo:
     *  - 0x02 ou 0x03
     */
    fun publicKeyFromPrivate(privateKey: ByteArray): ByteArray {
        require(privateKey.size == 32) {
            "Private key must be 32 bytes"
        }

        val curveParams = ECNamedCurveTable.getParameterSpec("secp256k1")
        val g = curveParams.g
        val n = curveParams.n

        val d = BigInteger(1, privateKey)

        require(d > BigInteger.ZERO && d < n) {
            "Private key out of range"
        }

        val q: ECPoint = g.multiply(d).normalize()
        return q.getEncoded(true)
    }

    /**
     * Retorna pubkey x-only (32 bytes) — Taproot
     */
    fun xOnlyPublicKeyFromPrivate(privateKey: ByteArray): ByteArray {
        val full = publicKeyFromPrivate(privateKey)
        return full.copyOfRange(1, 33)
    }

    // =================================================
    // Assinatura ECDSA — DER + LOW-S (EXISTENTE)
    // =================================================

    /**
     * Assina um hash de 32 bytes (double-SHA256)
     *
     * Retorna:
     *  - Assinatura DER (BIP-66)
     *  - LOW-S normalizado (BIP-62)
     *  - NÃO inclui o sighash byte
     */
    fun sign(
        privateKey: ByteArray,
        messageHash: ByteArray
    ): ByteArray {

        require(privateKey.size == 32) {
            "Private key must be 32 bytes"
        }

        require(messageHash.size == 32) {
            "Message hash must be 32 bytes"
        }

        val d = BigInteger(1, privateKey)
        val privKeyParams = ECPrivateKeyParameters(d, DOMAIN)

        val signer = ECDSASigner(
            HMacDSAKCalculator(SHA256Digest())
        )

        signer.init(true, privKeyParams)

        val (rRaw, sRaw) = signer.generateSignature(messageHash)

        // LOW-S enforcement (Bitcoin rule)
        val s = if (sRaw > HALF_CURVE_ORDER) {
            CURVE.n.subtract(sRaw)
        } else {
            sRaw
        }

        return derEncode(rRaw, s)
    }

    // =================================================
    // Assinatura Schnorr — BIP340 (NOVO)
    // =================================================

    /**
     * Assinatura Schnorr (Taproot)
     *
     * Retorna:
     *  - 64 bytes fixos (r || s)
     *  - NÃO DER
     *  - NÃO inclui sighash byte
     */
    fun signSchnorr(
        privateKey: ByteArray,
        messageHash: ByteArray
    ): ByteArray {

        require(privateKey.size == 32) {
            "Private key must be 32 bytes"
        }

        require(messageHash.size == 32) {
            "Message hash must be 32 bytes"
        }

        val d = BigInteger(1, privateKey)
        val privKeyParams = ECPrivateKeyParameters(d, DOMAIN)

        val signer = SchnorrSigner()
        signer.init(true, privKeyParams)

        signer.update(messageHash, 0, messageHash.size)
        return signer.generateSignature()
    }

    // =================================================
    // DER encoding manual (BIP-66)
    // =================================================

    private fun derEncode(r: BigInteger, s: BigInteger): ByteArray {
        val rEnc = encodeDerInt(r)
        val sEnc = encodeDerInt(s)

        val out = ByteArrayOutputStream()
        out.write(0x30) // SEQUENCE
        out.write(rEnc.size + sEnc.size)
        out.write(rEnc)
        out.write(sEnc)

        return out.toByteArray()
    }

    private fun encodeDerInt(value: BigInteger): ByteArray {
        var bytes = value.toByteArray()

        // remove zero desnecessário
        if (bytes.size > 1 && bytes[0] == 0x00.toByte() && bytes[1] < 0x80.toByte()) {
            bytes = bytes.copyOfRange(1, bytes.size)
        }

        // força inteiro positivo
        if (bytes[0] >= 0x80.toByte()) {
            bytes = byteArrayOf(0x00) + bytes
        }

        return byteArrayOf(
            0x02,                 // INTEGER
            bytes.size.toByte()
        ) + bytes
    }
}
