package com.pokewallet.crypto

object KeyDerivation {

    fun derive(
        seed: ByteArray,
        path: IntArray
    ): HDKey {

        var key = Bip32.fromSeed(seed)

        for (index in path) {
            val hardened = index and 0x80000000.toInt() != 0
            val idx = index and 0x7fffffff

            key = Bip32.deriveChild(
                parent = key,
                index = idx,
                hardened = hardened
            )
        }

        return key
    }
}



package com.pokewallet.crypto

object KeyDerivation {

    fun masterFromSeed(seed: ByteArray): HDKey =
        Bip32.fromSeed(seed)

    fun account(
        seed: ByteArray,
        walletType: WalletType,
        coinType: Int = 0,
        account: Int = 0
    ): HDKey {

        val root = masterFromSeed(seed)

        val path = "m/${walletType.purpose}'/$coinType'/$account'"

        return DerivationPath.derive(root, path)
    }

    fun externalAddressKey(
        accountKey: HDKey,
        index: Int
    ): HDKey =
        Bip32.deriveChild(
            parent = Bip32.deriveChild(accountKey, 0, false),
            index = index,
            hardened = false
        )
}
